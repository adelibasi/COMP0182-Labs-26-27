cmake_minimum_required(VERSION 3.8)
project(lab6_planner)
find_package(ament_cmake REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(lab6_planner_interfaces REQUIRED)
add_library(${PROJECT_NAME} SHARED src/student_planner.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
ament_target_dependencies(${PROJECT_NAME} pluginlib rclcpp nav2_core nav2_costmap_2d
  nav2_msgs nav_msgs lab6_planner_interfaces)
pluginlib_export_plugin_description_file(nav2_core plugin.xml)
install(TARGETS ${PROJECT_NAME} DESTINATION lib)
ament_package()
